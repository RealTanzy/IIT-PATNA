\begin{frame}{A* Search: Why \& How}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\includegraphics[width=\textwidth]{slide_3_astar_admissibility.png}
\end{column}

\begin{column}{0.48\textwidth}
\scriptsize
\textbf{Why A* (not ToT, not MCTS)?}
\begin{itemize}\setlength{\itemsep}{0pt}\tiny
    \item Isolates \textbf{branching} from evaluator design
    \item Clean separation: frontier + cost + heuristic
    \item ToT entangles evaluator; MCTS adds rollout noise
\end{itemize}

\vspace{0.3em}
\textbf{Formulation:}\\[0.2em]
\tiny
$f(n) = g(n) + h(n)$\\
$g(n) = d_n + \sum(1-c_t)$ \quad {\color{gray}(length + uncertainty)}\\
$h(n) = \omega_d(H_{\min}-d_n) + \omega_c(1-E(n))$\\[0.3em]

\scriptsize
\textbf{Admissibility (proven):}
\begin{itemize}\setlength{\itemsep}{0pt}\tiny
    \item $h(n) \leq h^*(n)$ for all non-goal nodes
    \item Condition: $\omega_d + \omega_c \leq c_{\min}$
    \item Our values: $0.3 + 0.3 = 0.6 < 1.01$ \textcolor{green!60!black}{$\checkmark$}
    \item \textbf{Guarantees optimal path under budget}
\end{itemize}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Topology Features}

\begin{center}
\scriptsize
\begin{tabular}{c|c}
\textcolor{red!70!black}{\textbf{Route to A* (search)}} & \textcolor{blue!70!black}{\textbf{Route to CoT (linear)}} \\
\hline
High branching coefficient & Low branching coefficient \\
Multiple hops / entities & Short, single-step question \\
Many connectives (if, because) & Direct factual recall \\
Low linearity score & High linearity score \\
\end{tabular}
\end{center}

\vspace{0.2em}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\scriptsize
\begin{enumerate}\setlength{\itemsep}{0pt}
    \item Question length
    \item Context length
    \item \colorbox{yellow!30}{Logical connectives} $\star$
    \item Negation markers
    \item \colorbox{yellow!30}{Multi-hop indicators} $\star$
    \item Comparison words
    \item Sub-question marks
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\scriptsize
\begin{enumerate}\setlength{\itemsep}{0pt}\setcounter{enumi}{7}
    \item \colorbox{yellow!30}{Context sentences} $\star$
    \item Named entities
    \item Linearity score
    \item Answer options
    \item Depth estimate
    \item \colorbox{yellow!30}{Branching complexity} $\star$
\end{enumerate}

\vspace{0.2em}
\small
\colorbox{green!20}{$\text{BC} = \frac{\text{ctx\_sents} \times \text{hop}}{\text{q\_len}}$} \quad {\tiny($R^2 = 0.94$)}
\end{column}
\end{columns}
{\tiny $\star$ = top XGBoost importance. All features from input text only — zero LLM inference.}

\end{frame}

\tiny
\begin{tabular}{lcc}
\toprule
Dataset & CoT-only & A*-only \\
\midrule
StrategyQA & 14.3\% & 10.8\% \\
HotpotQA & 17.2\% & 17.4\% \\
CodeContests & 9.2\% & 18.8\% \\
\bottomrule
\end{tabular}\\[0.2em]
\scriptsize Oracle gap: up to \textbf{14pp}
\end{column}

\begin{column}{0.52\textwidth}
\scriptsize
\textbf{How the router leverages this:}
\begin{itemize}\setlength{\itemsep}{1pt}\tiny
    \item Each instance: features predict $\rightarrow$ CoT or A*
    \item ``Branching'' problems $\rightarrow$ A* (multi-hop, algorithmic)
    \item ``Linear'' problems $\rightarrow$ CoT (factual, single-step)
    \item Router recovers \textbf{40\%} of the oracle gap
\end{itemize}

\vspace{0.3em}
\scriptsize
\textbf{Practical benefit:}
\begin{itemize}\setlength{\itemsep}{1pt}\tiny
    \item A* costs 24$\times$ CoT
    \item Router applies A* only when needed ($\sim$30\% of instances)
    \item Effective cost: $\sim$8$\times$ CoT (not 24$\times$)
    \item \textbf{Best of both worlds}: high accuracy, manageable cost
\end{itemize}

\vspace{0.2em}
\centering
\fbox{\tiny Neither method alone is optimal $\rightarrow$ topology routing fills the gap}
\end{column}
\end{columns}

\end{frame}
